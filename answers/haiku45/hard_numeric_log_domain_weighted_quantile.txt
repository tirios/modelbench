import numpy as np

def log_sum_exp(xs):
    """Compute log(sum(exp(x))) numerically stably."""
    arr = np.asarray(xs, dtype=float).flatten()

    # Check for NaN or +inf
    if np.any(np.isnan(arr)) or np.any(np.isinf(arr) & (arr > 0)):
        raise ValueError()

    # If empty or all -inf, return -inf
    if len(arr) == 0 or np.all(np.isinf(arr)):
        return float('-inf')

    # Remove -inf entries (they contribute 0 to the sum)
    finite = arr[arr != float('-inf')]

    if len(finite) == 0:
        return float('-inf')

    # Use the log-sum-exp trick
    max_val = np.max(finite)
    result = max_val + np.log(np.sum(np.exp(finite - max_val)))

    return float(result)


def weighted_quantile(values, log_weights, qs):
    """Compute weighted quantiles."""
    values = np.asarray(values, dtype=float).flatten()
    log_weights = np.asarray(log_weights, dtype=float).flatten()
    qs = np.asarray(qs, dtype=float).flatten()

    n = len(values)

    # V1: Length check
    if len(log_weights) != n:
        raise ValueError()

    # V2: Check values for NaN, +inf, -inf
    if np.any(np.isnan(values)) or np.any(np.isinf(values)):
        raise ValueError()

    # V3: Check log_weights for NaN or +inf
    if np.any(np.isnan(log_weights)) or np.any(np.isinf(log_weights) & (log_weights > 0)):
        raise ValueError()

    # V4: Check qs for NaN or out of [0, 1]
    if np.any(np.isnan(qs)) or np.any(qs < 0) or np.any(qs > 1):
        raise ValueError()

    # V5: Check if all weights are -inf or n == 0
    if n == 0 or np.all(log_weights == float('-inf')):
        raise ValueError()

    # If qs is empty, return empty list
    if len(qs) == 0:
        return []

    # A1: Discard -inf weights
    mask = log_weights != float('-inf')
    kept_values = values[mask]
    kept_log_weights = log_weights[mask]

    # A2: Compute t_i = exp(lw_i - L)
    L = np.max(kept_log_weights)
    t_i = np.exp(kept_log_weights - L)

    # Discard entries where t_i == 0
    nonzero_mask = t_i != 0.0
    kept_values = kept_values[nonzero_mask]
    t_i = t_i[nonzero_mask]

    # A3: Normalize to get weights
    Z = np.sum(t_i)
    w_i = t_i / Z

    # A4: Merge equal values
    unique_vals, indices = np.unique(kept_values, return_inverse=True)
    v_k = np.zeros(len(unique_vals))
    for i, idx in enumerate(indices):
        v_k[idx] += w_i[i]

    # A5: If only one distinct value
    if len(unique_vals) == 1:
        return [float(unique_vals[0])] * len(qs)

    # A6: Compute cumulative and rescale
    m = len(unique_vals)
    C_k = np.cumsum(v_k)
    p_k = C_k - v_k / 2

    h_k = (p_k - p_k[0]) / (p_k[-1] - p_k[0])

    # A7: Interpolate for each q
    result = []
    for q in qs:
        if q <= 0:
            result.append(float(unique_vals[0]))
        elif q >= 1:
            result.append(float(unique_vals[-1]))
        else:
            # Find k such that h_k <= q <= h_{k+1}
            for k in range(m - 1):
                if h_k[k] <= q <= h_k[k + 1]:
                    u_k = unique_vals[k]
                    u_kp1 = unique_vals[k + 1]
                    val = u_k + (q - h_k[k]) * (u_kp1 - u_k) / (h_k[k + 1] - h_k[k])
                    result.append(float(val))
                    break

    return result
