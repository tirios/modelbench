def min_points(intervals):
    if not intervals:
        return 0

    # Sort intervals by their end point
    intervals.sort(key=lambda x: x[1])

    count = 1
    last_point = intervals[0][1]  # Place first point at end of first interval

    for start, end in intervals[1:]:
        # If last_point is not in this interval [start, end]
        if last_point < start:
            # Place a new point at the end of this interval
            count += 1
            last_point = end

    return count
