import numpy as np

def fit_harmonics(theta, y, K):
    theta = np.asarray(theta)
    y = np.asarray(y)

    n = len(theta)
    X = np.ones((n, 2*K + 1))

    # Cosine columns for k = 1..K
    for k in range(1, K+1):
        X[:, k] = np.cos(k * theta)

    # Sine columns for k = 1..K
    for k in range(1, K+1):
        X[:, K + k] = np.sin(k * theta)

    # Solve least squares
    coef, _, _, _ = np.linalg.lstsq(X, y, rcond=None)

    a0 = coef[0]
    a_list = list(coef[1:K+1])
    b_list = list(coef[K+1:2*K+1])

    return (a0, a_list, b_list)
