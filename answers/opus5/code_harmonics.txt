```python
import numpy as np


def fit_harmonics(theta, y, K):
    """Least-squares fit of a truncated Fourier series to (theta, y).

    Model: y ~= a0 + sum_{k=1..K} [ a_k*cos(k*theta) + b_k*sin(k*theta) ]

    Returns (a0, [a_1..a_K], [b_1..b_K]).
    """
    theta = np.asarray(theta, dtype=float).ravel()
    y = np.asarray(y, dtype=float).ravel()

    if theta.shape[0] != y.shape[0]:
        raise ValueError("theta and y must have the same number of samples")

    K = int(K)
    if K < 0:
        raise ValueError("K must be non-negative")

    n = theta.shape[0]

    # Design matrix: [1, cos(theta), sin(theta), ..., cos(K*theta), sin(K*theta)]
    X = np.empty((n, 1 + 2 * K), dtype=float)
    X[:, 0] = 1.0
    for k in range(1, K + 1):
        kt = k * theta
        X[:, 2 * k - 1] = np.cos(kt)
        X[:, 2 * k] = np.sin(kt)

    # lstsq handles rank-deficient designs (e.g. too few or aliased samples)
    # via the minimum-norm solution instead of raising.
    coef, _, _, _ = np.linalg.lstsq(X, y, rcond=None)

    a0 = float(coef[0])
    a_list = [float(coef[2 * k - 1]) for k in range(1, K + 1)]
    b_list = [float(coef[2 * k]) for k in range(1, K + 1)]

    return a0, a_list, b_list
```
