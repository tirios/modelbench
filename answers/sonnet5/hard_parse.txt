```python
import re

_MONTH_NAMES = {
    'jan': 1, 'feb': 2, 'mar': 3, 'apr': 4, 'may': 5, 'jun': 6,
    'jul': 7, 'aug': 8, 'sep': 9, 'oct': 10, 'nov': 11, 'dec': 12,
}

_RE_ISO = re.compile(r'^([0-9]{4})-([0-9]{2})-([0-9]{2})$')
_RE_SLASH = re.compile(r'^([0-9]{1,2})/([0-9]{1,2})/([0-9]{4})$')
_RE_MON = re.compile(r'^([0-9]{1,2}) +([A-Za-z]{3}) +([0-9]{4})$')


def _is_leap_year(y):
    return y % 4 == 0 and (y % 100 != 0 or y % 400 == 0)


def _days_in_month(y, m):
    lengths = [31, 29 if _is_leap_year(y) else 28, 31, 30, 31, 30,
               31, 31, 30, 31, 30, 31]
    return lengths[m - 1]


def _is_valid_calendar_date(y, m, d):
    if not (1000 <= y <= 9999):
        return False
    if not (1 <= m <= 12):
        return False
    if d < 1:
        return False
    return d <= _days_in_month(y, m)


def _parse_date(field):
    s = field.strip(' \t')
    if not s:
        return None

    m = _RE_ISO.match(s)
    if m:
        y, mo, d = int(m.group(1)), int(m.group(2)), int(m.group(3))
        if _is_valid_calendar_date(y, mo, d):
            return '%04d-%02d-%02d' % (y, mo, d)
        return None

    m = _RE_SLASH.match(s)
    if m:
        a, b, y = int(m.group(1)), int(m.group(2)), int(m.group(3))
        if _is_valid_calendar_date(y, b, a):
            return '%04d-%02d-%02d' % (y, b, a)
        if _is_valid_calendar_date(y, a, b):
            return '%04d-%02d-%02d' % (y, a, b)
        return None

    m = _RE_MON.match(s)
    if m:
        d = int(m.group(1))
        mon_key = m.group(2).lower()
        if mon_key not in _MONTH_NAMES:
            return None
        mo = _MONTH_NAMES[mon_key]
        y = int(m.group(3))
        if _is_valid_calendar_date(y, mo, d):
            return '%04d-%02d-%02d' % (y, mo, d)
        return None

    return None


def _commas_are_grouping(magnitude):
    n = len(magnitude)
    for idx, ch in enumerate(magnitude):
        if ch != ',':
            continue
        nxt = magnitude[idx + 1: idx + 4]
        if len(nxt) != 3 or not nxt.isdigit():
            return False
        after = idx + 4
        if after < n and magnitude[after] != ',':
            return False
    return True


def _parse_amount(field):
    s = field.strip(' \t')
    if s == '':
        return None

    negative = False
    paren = False
    if len(s) >= 2 and s[0] == '(' and s[-1] == ')':
        s = s[1:-1].strip(' \t')
        paren = True
        negative = True
    elif s.startswith('(') or s.endswith(')'):
        return None

    if s.startswith('+') or s.startswith('-'):
        if paren:
            return None
        if s[0] == '-':
            negative = True
        s = s[1:]

    if s.startswith('$'):
        s = s[1:]

    percent = False
    if s.endswith('%'):
        percent = True
        s = s[:-1]

    magnitude = s
    if magnitude == '':
        return None
    if not all(ch in '0123456789,.' for ch in magnitude):
        return None

    has_comma = ',' in magnitude
    has_dot = '.' in magnitude

    decimal_sep = None
    group_sep = None

    if has_comma and has_dot:
        last_comma = magnitude.rfind(',')
        last_dot = magnitude.rfind('.')
        if last_dot > last_comma:
            decimal_sep = '.'
            group_sep = ','
        else:
            decimal_sep = ','
            group_sep = '.'
    elif has_comma:
        if _commas_are_grouping(magnitude):
            group_sep = ','
        else:
            decimal_sep = ','
    elif has_dot:
        if magnitude.count('.') >= 2:
            group_sep = '.'
        else:
            decimal_sep = '.'

    if decimal_sep is not None:
        parts = magnitude.split(decimal_sep)
        if len(parts) != 2:
            return None
        int_part, frac_part = parts
        if frac_part == '' or not frac_part.isdigit():
            return None
    else:
        int_part = magnitude
        frac_part = ''

    if group_sep is not None:
        groups = int_part.split(group_sep)
        if len(groups) < 2:
            return None
        if not (1 <= len(groups[0]) <= 3 and groups[0].isdigit()):
            return None
        for g in groups[1:]:
            if len(g) != 3 or not g.isdigit():
                return None
        int_digits = ''.join(groups)
    else:
        if int_part == '' or not int_part.isdigit():
            return None
        int_digits = int_part

    value = float(int_digits + '.' + (frac_part if frac_part != '' else '0'))

    if percent:
        value = value / 100.0
    if negative:
        value = -value

    return value


def _scan_fields(line):
    fields = []
    n = len(line)
    pos = 0
    while True:
        j = pos
        while j < n and line[j] in (' ', '\t'):
            j += 1
        if j < n and line[j] == '"':
            j += 1
            chars = []
            closed = False
            while j < n:
                c = line[j]
                if c == '"':
                    if j + 1 < n and line[j + 1] == '"':
                        chars.append('"')
                        j += 2
                        continue
                    else:
                        j += 1
                        closed = True
                        break
                else:
                    chars.append(c)
                    j += 1
            if not closed:
                return None, False
            while j < n and line[j] in (' ', '\t'):
                j += 1
            if j < n and line[j] == ',':
                fields.append(''.join(chars))
                pos = j + 1
                continue
            elif j == n:
                fields.append(''.join(chars))
                return fields, True
            else:
                return None, False
        else:
            k = j
            while k < n and line[k] != ',':
                k += 1
            value = line[j:k].strip(' \t')
            fields.append(value)
            if k < n:
                pos = k + 1
                continue
            else:
                return fields, True


def parse_records(text):
    if not isinstance(text, str):
        raise TypeError('text must be a str')

    records = []
    errors = []
    id_to_index = {}

    lines = text.split('\n')
    for i, raw_line in enumerate(lines):
        line_number = i + 1
        line = raw_line
        if line.endswith('\r'):
            line = line[:-1]

        if line.strip(' \t') == '':
            continue

        if line.lstrip(' \t').startswith('#'):
            continue

        fields, ok = _scan_fields(line)
        if not ok:
            errors.append((line_number, 'BAD_QUOTE'))
            continue

        if len(fields) not in (3, 4):
            errors.append((line_number, 'FIELD_COUNT'))
            continue

        id_value = fields[0]
        if id_value.strip(' \t') == '':
            errors.append((line_number, 'BAD_ID'))
            continue

        date_value = _parse_date(fields[1])
        if date_value is None:
            errors.append((line_number, 'BAD_DATE'))
            continue

        amount_value = _parse_amount(fields[2])
        if amount_value is None:
            errors.append((line_number, 'BAD_AMOUNT'))
            continue

        note_value = fields[3] if len(fields) == 4 else ''

        record = {
            'id': id_value,
            'date': date_value,
            'amount': amount_value,
            'note': note_value,
        }

        if id_value in id_to_index:
            records[id_to_index[id_value]] = record
        else:
            id_to_index[id_value] = len(records)
            records.append(record)

    return records, errors
```
