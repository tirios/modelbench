def min_points(intervals):
    if not intervals:
        return 0

    sorted_intervals = sorted(intervals, key=lambda iv: iv[1])

    count = 0
    last_point = None
    for start, end in sorted_intervals:
        if last_point is None or start > last_point:
            count += 1
            last_point = end

    return count
